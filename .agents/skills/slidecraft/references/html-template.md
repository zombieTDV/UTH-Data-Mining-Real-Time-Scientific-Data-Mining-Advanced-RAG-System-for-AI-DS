# HTML Presentation Template

Reference architecture for generating slide presentations. Every presentation follows this structure.

## Base HTML Structure

```html
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Presentation Title</title>

    <!-- Fonts: Google Fonts or Fontshare — never system fonts -->
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link href="https://fonts.googleapis.com/css2?family=..." rel="stylesheet" />

    <style>
      /* ===========================================
           CSS CUSTOM PROPERTIES (THEME)
           Change these `:root` vars to retheme the whole deck.
           =========================================== */
      :root {
        /* Colors — from chosen style preset */
        --bg-primary: #0a0f1c;
        --bg-secondary: #111827;
        --text-primary: #ffffff;
        --text-secondary: #9ca3af;
        --accent: #00ffcc;
        --accent-glow: rgba(0, 255, 204, 0.3);

        /* Typography — MUST use clamp() */
        --font-display: "Clash Display", sans-serif;
        --font-body: "Satoshi", sans-serif;
        --title-size: clamp(2rem, 6vw, 5rem);
        --subtitle-size: clamp(0.875rem, 2vw, 1.25rem);
        --body-size: clamp(0.75rem, 1.2vw, 1rem);

        /* Spacing — MUST use clamp() */
        --slide-padding: clamp(1.5rem, 4vw, 4rem);
        --content-gap: clamp(1rem, 2vw, 2rem);

        /* Animation */
        --ease-out-expo: cubic-bezier(0.16, 1, 0.3, 1);
        --duration-normal: 0.6s;
      }

      * { margin: 0; padding: 0; box-sizing: border-box; }

      /* --- PASTE FULL viewport-base.css CONTENTS HERE --- */

      /* ===========================================
           ANIMATIONS
           =========================================== */
      .reveal {
        opacity: 0;
        transform: translateY(30px);
        transition: opacity var(--duration-normal) var(--ease-out-expo),
                    transform var(--duration-normal) var(--ease-out-expo);
      }
      .slide.visible .reveal { opacity: 1; transform: translateY(0); }

      .reveal:nth-child(1) { transition-delay: 0.1s; }
      .reveal:nth-child(2) { transition-delay: 0.2s; }
      .reveal:nth-child(3) { transition-delay: 0.3s; }
      .reveal:nth-child(4) { transition-delay: 0.4s; }

      /* ... preset-specific styles below ... */
    </style>
  </head>
  <body>
    <!-- Progress bar -->
    <div class="progress-bar"></div>

    <!-- Navigation dots -->
    <nav class="nav-dots"><!-- Generated by JS --></nav>

    <!-- === TITLE SLIDE === -->
    <section class="slide title-slide">
      <div class="slide-content">
        <h1 class="reveal">Presentation Title</h1>
        <p class="reveal">Subtitle or author</p>
      </div>
    </section>

    <!-- === CONTENT SLIDE === -->
    <section class="slide">
      <div class="slide-content">
        <h2 class="reveal">Slide Title</h2>
        <p class="reveal">Content paragraph...</p>
        <ul class="bullet-list">
          <li class="reveal">Point one</li>
          <li class="reveal">Point two</li>
        </ul>
      </div>
    </section>

    <!-- More slides... -->

    <script>
      /* ===========================================
           SLIDE PRESENTATION CONTROLLER
           Handles navigation, animations, progress.
           =========================================== */
      class SlidePresentation {
        constructor() {
          this.slides = document.querySelectorAll(".slide");
          this.currentSlide = 0;
          this.isScrolling = false;
          this.setupIntersectionObserver();
          this.setupKeyboardNav();
          this.setupTouchNav();
          this.setupWheelNav();
          this.setupProgressBar();
          this.setupNavDots();
        }

        /* --- Intersection Observer: triggers .visible for CSS animations --- */
        setupIntersectionObserver() {
          const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
              if (entry.isIntersecting) {
                entry.target.classList.add('visible');
                const idx = Array.from(this.slides).indexOf(entry.target);
                if (idx !== -1) this.currentSlide = idx;
                this.updateNavDots();
                this.updateProgressBar();
              }
            });
          }, { threshold: 0.5 });
          this.slides.forEach(slide => observer.observe(slide));
        }

        /* --- Keyboard: arrows, space, page up/down --- */
        setupKeyboardNav() {
          document.addEventListener('keydown', (e) => {
            if (e.target.getAttribute && e.target.getAttribute('contenteditable')) return;
            if (['ArrowDown','ArrowRight',' ','PageDown'].includes(e.key)) {
              e.preventDefault();
              this.goToSlide(this.currentSlide + 1);
            } else if (['ArrowUp','ArrowLeft','PageUp'].includes(e.key)) {
              e.preventDefault();
              this.goToSlide(this.currentSlide - 1);
            } else if (e.key === 'Home') {
              e.preventDefault();
              this.goToSlide(0);
            } else if (e.key === 'End') {
              e.preventDefault();
              this.goToSlide(this.slides.length - 1);
            }
          });
        }

        /* --- Touch: swipe up/down --- */
        setupTouchNav() {
          let startY = 0, startX = 0;
          document.addEventListener('touchstart', e => {
            startY = e.touches[0].clientY;
            startX = e.touches[0].clientX;
          });
          document.addEventListener('touchend', e => {
            const dy = startY - e.changedTouches[0].clientY;
            const dx = startX - e.changedTouches[0].clientX;
            // Only navigate if vertical swipe is dominant
            if (Math.abs(dy) > Math.abs(dx) && Math.abs(dy) > 50) {
              this.goToSlide(this.currentSlide + (dy > 0 ? 1 : -1));
            }
          });
        }

        /* --- Mouse wheel with debounce --- */
        setupWheelNav() {
          document.addEventListener('wheel', (e) => {
            if (this.isScrolling) return;
            this.isScrolling = true;
            if (e.deltaY > 0) this.goToSlide(this.currentSlide + 1);
            else if (e.deltaY < 0) this.goToSlide(this.currentSlide - 1);
            setTimeout(() => { this.isScrolling = false; }, 800);
          }, { passive: true });
        }

        /* --- Progress bar --- */
        setupProgressBar() {
          this.progressBar = document.querySelector('.progress-bar');
          if (this.progressBar) {
            Object.assign(this.progressBar.style, {
              position: 'fixed', top: '0', left: '0', height: '3px',
              background: 'var(--accent, #00ffcc)', zIndex: '9999',
              transition: 'width 0.3s ease'
            });
          }
          this.updateProgressBar();
        }

        /* --- Nav dots --- */
        setupNavDots() {
          this.navDotsContainer = document.querySelector('.nav-dots');
          if (!this.navDotsContainer) return;
          // Always clear to prevent duplication on re-open
          this.navDotsContainer.innerHTML = "";
          this.slides.forEach((_, i) => {
            const dot = document.createElement('button');
            dot.classList.add('nav-dot');
            dot.setAttribute('aria-label', `Go to slide ${i + 1}`);
            dot.addEventListener('click', () => this.goToSlide(i));
            this.navDotsContainer.appendChild(dot);
          });
          this.updateNavDots();
        }

        goToSlide(index) {
          if (index < 0 || index >= this.slides.length) return;
          this.slides[index].scrollIntoView({ behavior: 'smooth' });
        }

        updateNavDots() {
          if (!this.navDotsContainer) return;
          this.navDotsContainer.querySelectorAll('.nav-dot').forEach((dot, i) => {
            dot.classList.toggle('active', i === this.currentSlide);
          });
        }

        updateProgressBar() {
          if (!this.progressBar) return;
          const pct = ((this.currentSlide + 1) / this.slides.length) * 100;
          this.progressBar.style.width = pct + '%';
        }
      }

      document.addEventListener('DOMContentLoaded', () => new SlidePresentation());
    </script>
  </body>
</html>
```

## Slide Type Patterns

Use a variety of slide layouts — not just bullet lists. Here are patterns to draw from:

### Title / Section Divider
```html
<section class="slide title-slide">
  <div class="slide-content" style="text-align: center;">
    <p class="reveal section-label">01</p>
    <h1 class="reveal">Section Title</h1>
    <p class="reveal subtitle">A brief tagline or subtitle</p>
  </div>
</section>
```

### Feature Grid (2×3)
```html
<section class="slide">
  <div class="slide-content">
    <h2 class="reveal">Key Features</h2>
    <div class="grid" style="grid-template-columns: repeat(3, 1fr);">
      <div class="card reveal" style="--i:0">
        <span class="card-icon">⚡</span>
        <h3>Feature One</h3>
        <p>Brief description</p>
      </div>
      <!-- ... up to 6 cards -->
    </div>
  </div>
</section>
```

### Two-Column Comparison
```html
<section class="slide">
  <div class="slide-content">
    <h2 class="reveal">Before vs After</h2>
    <div style="display: grid; grid-template-columns: 1fr 1fr; gap: var(--content-gap);">
      <div class="reveal-left">
        <h3>Before</h3>
        <ul class="bullet-list"><li>Item</li></ul>
      </div>
      <div class="reveal-right">
        <h3>After</h3>
        <ul class="bullet-list"><li>Item</li></ul>
      </div>
    </div>
  </div>
</section>
```

### Metric Highlight
```html
<section class="slide">
  <div class="slide-content" style="text-align: center;">
    <h2 class="reveal">By the Numbers</h2>
    <div style="display: flex; justify-content: space-around; margin-top: var(--content-gap);">
      <div class="reveal">
        <span class="metric-number" data-target="500">0</span>
        <span class="metric-label">Users</span>
      </div>
      <div class="reveal">
        <span class="metric-number" data-target="99">0</span>
        <span class="metric-label">% Uptime</span>
      </div>
    </div>
  </div>
</section>
```

### Quote / Callout
```html
<section class="slide">
  <div class="slide-content" style="display: flex; align-items: center; justify-content: center;">
    <blockquote class="reveal" style="max-width: 70%; text-align: center;">
      <p style="font-size: var(--h2-size); font-style: italic; line-height: 1.4;">
        "A great quote goes here — keep it to three lines maximum."
      </p>
      <cite class="reveal" style="margin-top: 1em; display: block;">— Attribution</cite>
    </blockquote>
  </div>
</section>
```

### Code Slide
```html
<section class="slide">
  <div class="slide-content">
    <h2 class="reveal">Code Example</h2>
    <pre class="reveal code-block"><code>function hello() {

```

## Inline Editing (Opt-In Only)

Only include if the user requested it. Uses JS-based hover — do NOT use CSS `~` sibling selector.

**Key components:**
- Edit hotzone (fixed 80×80px, top-left corner, invisible)
- Toggle button with 400ms grace period on mouseleave
- `E` key shortcut (skip when contenteditable is focused)
- `Ctrl+S` to export — must strip edit state from outerHTML before saving

**CRITICAL: `exportFile()` must:**
1. Remove `contenteditable` from all elements
2. Remove `body.edit-active` class
3. Remove `.active`/`.show` from toggle button and banner
4. Capture `outerHTML`
5. Restore all state so user can keep editing

## Image Handling

- User images → base64-encode inline: `<img src="data:image/png;base64,...">`
- No images → CSS visuals (gradients, shapes, patterns)
- All images: `max-height: min(50vh, 400px); object-fit: contain;`
- Never repeat same image on multiple slides (except logo on title + closing)
- Adapt border/shadow colors to match preset's accent

## Code Quality Checklist

- [ ] Semantic HTML (`<section>`, `<nav>`, `<main>`)
- [ ] Full keyboard accessibility (arrows, space, home, end)
- [ ] ARIA labels on interactive elements
- [ ] `prefers-reduced-motion` support (from viewport-base.css)
- [ ] Comments on every section: `/* === SECTION NAME === */`
- [ ] CSS variables in `:root` for easy theming
- [ ] All sizes use `clamp()` — no fixed px/rem alone
- [ ] `.slide` has `overflow: hidden` — no scrolling ever
